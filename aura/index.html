// Tooltip-ai Aura: Detección anticipada de descanso de mouse sin eventos de clic
public void OnMouseDwellCompleted(Point cursorPosition) {
    // 1. Obtener identificador de ventana bajo el cursor en 0.2ms
    IntPtr hWnd = NativeMethods.WindowFromPoint(cursorPosition);
    AutomationElement target = AutomationElement.FromHandle(hWnd);
    
    // 2. Evaluar árbol de accesibilidad y telemetría de riesgo
    RiskAssessment risk = CognitiveEvaluator.EvaluateOperationalSafety(target);
    
    // 3. Proyectar HUD translúcido con efecto Mica de Windows 11
    AcrylicHudWindow.ShowOverlay(target, risk, backdrop: BackdropType.Mica);
}
