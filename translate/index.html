// Tooltip-ai Translate v1.1.2: Auto-detección multilingüe — sin configurar idioma origen
public async Task<TranslationReport> TranslateScreenRegionAsync(Direct3DSurface surface, Rect selection) {
    // 1. Convertir buffer D3D a SoftwareBitmap en memoria GPU compartida
    using var softwareBitmap = SoftwareBitmap.CreateCopyFromSurface(surface);

    // 2. OCR sin idioma fijo: Windows.Media.Ocr detecta el script automáticamente
    //    Soporta: Japonés, Chino, Coreano, Árabe, Cirílico, Latino y 40+ más
    var ocrEngine = OcrEngine.TryCreateFromUserProfileLanguages();
    var ocrResult = await ocrEngine.RecognizeAsync(softwareBitmap);

    // 3. Detección de idioma mediante Azure Cognitive Services (sin upload de imagen)
    string detectedLang = await LanguageDetector.DetectAsync(ocrResult.Text);
    // → "ja", "de", "zh", "en", "ko", "ar"... cualquier idioma, automáticamente

    // 4. Proteger campos de contraseña: texto descartado si está en campo seguro
    if (SecurityAudit.IsPasswordFieldUnderCursor(selection))
        return TranslationReport.ProtectedField();

    // 5. Inferencia semántica con contexto gamer/técnico y preservación de nombres
    string translatedText = await ContextualBrain.TranslateAsync(
        ocrResult.Text, fromLang: detectedLang, toLang: UserSettings.TargetLanguage);
    return new TranslationReport(ocrResult.Text, translatedText, detectedLang);
}
