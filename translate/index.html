// Tooltip-ai Translate: Extracción visual de pantalla sin inyectar DLLs en el juego
public async Task<TranslationReport> TranslateScreenRegionAsync(Direct3DSurface surface, Rect selection) {
    // 1. Convertir buffer D3D a SoftwareBitmap en memoria compartida GPU
    using var softwareBitmap = SoftwareBitmap.CreateCopyFromSurface(surface);
    
    // 2. Extraer glifos con el motor OCR nativo de Windows (50+ idiomas)
    var ocrEngine = OcrEngine.TryCreateFromLanguage(new Language("ja-JP"));
    var ocrResult = await ocrEngine.RecognizeAsync(softwareBitmap);
    
    // 3. Aislar credenciales de acceso: Si hay un campo IsPassword el texto se descarta
    if (SecurityAudit.IsPasswordFieldUnderCursor(selection)) {
        return TranslationReport.ProtectedField();
    }
    
    // 4. Inferencia semántica multimodal con preservación de nombres propios
    string translatedText = await ContextualBrain.TranslateGamerJargonAsync(ocrResult.Text);
    return new TranslationReport(ocrResult.Text, translatedText, ocrResult.Lines);
}
