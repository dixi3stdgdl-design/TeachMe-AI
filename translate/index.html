// Tooltip-ai Translate: Extracción OCR directa sin dependencias externas
public async Task<OcrResult> RecognizeScreenAreaAsync(SoftwareBitmap bitmap) {
    // 1. Inicializar motor de OCR local de Windows 11
    OcrEngine engine = OcrEngine.TryCreateFromLanguage(new Language("ja-JP"));
    
    // 2. Extraer glifos con aceleración vectorial en CPU / Direct3D
    OcrResult ocrResult = await engine.RecognizeAsync(bitmap);
    
    // 3. Salvaguarda de privacidad: Evitar OCR si es campo de credencial
    if (AutomationElement.FocusedElement?.Current.IsPassword == true) {
        return OcrResult.FromMasked("[PROTECTED_PASSWORD_FIELD]");
    }
    
    return ocrResult;
}
