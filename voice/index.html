// Tooltip-ai Voice: Síntesis de voz directa sin buffering ni consumo de tokens cloud
public void SpeakVerdictAsync(string explanation) {
    using (var synth = new SpeechSynthesizer()) {
        // 1. Configurar voz neural en español de alta inteligibilidad
        synth.SelectVoiceByHints(VoiceGender.Female, VoiceAge.Adult, 0, new CultureInfo("es-ES"));
        synth.Rate = 1; // Velocidad calibrada para gaming
        
        // 2. Activar Audio Ducking en el mezclador de Windows
        AudioSessionController.ApplyDucking(level: 0.4f);
        
        // 3. Reproducir de forma asíncrona sin bloquear la interfaz
        synth.SpeakAsync(explanation);
    }
}
